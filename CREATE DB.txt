-- ==========================================
-- 0. RIMOZIONE TABELLE ESISTENTI
-- ==========================================
DROP TABLE IF EXISTS public.faq CASCADE;
DROP TABLE IF EXISTS public.prenotazioni CASCADE;
DROP TABLE IF EXISTS public.utenti CASCADE;
DROP TABLE IF EXISTS public.giochi CASCADE;

-- ==========================================
-- 1. CREAZIONE STRUTTURA
-- ==========================================

-- Tabella Giochi
CREATE TABLE public.giochi (
    nome_gioco character varying(50) PRIMARY KEY,
    immagine character varying(255)
);

-- Tabella Utenti
CREATE TABLE public.utenti (
    username character varying(30) PRIMARY KEY,
    nome_completo character varying(50) NOT NULL,
    email character varying(100) NOT NULL UNIQUE,
    pass text
);

-- Tabella Prenotazioni
CREATE TABLE public.prenotazioni (
    id_prenotazione SERIAL PRIMARY KEY,
    username_utente character varying(30) REFERENCES public.utenti(username) ON UPDATE CASCADE ON DELETE CASCADE,
    nome_gioco character varying(50) REFERENCES public.giochi(nome_gioco) ON UPDATE CASCADE,
    data_ora timestamp without time zone NOT NULL,
    numero_pista integer,
    numero_tavolo integer,
    numero_persone integer
);

-- Indici di unicità per evitare conflitti di prenotazione (Piste e Tavoli)
CREATE UNIQUE INDEX unica_prenotazione_pista ON public.prenotazioni (nome_gioco, data_ora, numero_pista) WHERE (numero_pista IS NOT NULL);
CREATE UNIQUE INDEX unica_prenotazione_tavolo ON public.prenotazioni (nome_gioco, data_ora, numero_tavolo) WHERE (numero_tavolo IS NOT NULL);

-- Tabella Recensioni (FAQ)
CREATE TABLE public.faq (
    id_recensione SERIAL PRIMARY KEY,
    username character varying(30) NOT NULL REFERENCES public.utenti(username) ON UPDATE CASCADE ON DELETE CASCADE,
    recensione character varying(300) NOT NULL,
    data_recensione timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
    stelle integer CHECK (stelle >= 1 AND stelle <= 5)
);

-- ==========================================
-- 2. PERMESSI ALLE TABELLE 
-- ==========================================
GRANT ALL PRIVILEGES ON TABLE utenti TO www;
GRANT ALL PRIVILEGES ON TABLE giochi TO www;
GRANT ALL PRIVILEGES ON TABLE prenotazioni TO www;
GRANT ALL PRIVILEGES ON TABLE faq TO www;
